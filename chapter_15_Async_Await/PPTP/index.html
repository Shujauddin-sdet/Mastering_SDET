142 - Async/Await Intro

142_Async_Await.js Syntax Sugar

async/await is syntactic sugar over Promises. Instead of .then() chains, you write code that looks synchronous using await to pause until each promise resolves.

// Promise chain (old way)
getToken()
  .then(function(token) { return getUser(token); })
  .then(function(user) { console.log(user); });

// Async/Await (new way) — same logic, cleaner!
async function run() {
    let token = await getToken();
    let user  = await getUser(token);
    console.log(user);
}

143 - Converted Code

143_Coverted_Code.js Promise → Await

Converting a real Promise chain (open browser → login → enter credentials) into clean async/await code. Shows the before/after side-by-side to demonstrate the readability improvement.

function openBrowser() {
    return new Promise(function(resolve) {
        resolve("Browser opened!");
    });
}

function goToLogin() {
    return new Promise(function(resolve) {
        resolve("Login page loaded");
    });
}

// Clean async/await version
async function runTest() {
    let step1 = await openBrowser();
    console.log(step1);
    let step2 = await goToLogin();
    console.log(step2);
}

144 - async Keyword

144_AA.js async Function

An async function ALWAYS returns a Promise, even if you return a plain value. You still use .then() to consume the result from outside.

async function getTestResults() {
    return "Pass";
}

// async function ALWAYS returns a Promise
getTestResults().then(function(result) {
    console.log(result); // "Pass"
});

145 - Try/Catch

145_Try_Catch.js Error Handling

With async/await, use try/catch for error handling instead of .catch(). The finally block runs regardless — mapping directly to .then()/.catch()/.finally().

async function testAPI() {
    try {
        let result = await Promise.reject("503 Service Unavailable");
        console.log("Result", result);
    } catch (error) {
        console.log("Error", error);
    } finally {
        console.log("Clean up!!");
    }
}

testAPI();

146 - Sequential Execution

146_Sequential_Execution.js One After Another

When Step 2 depends on Step 1's result, you MUST run them sequentially using await one after another. Each await pauses until the previous promise resolves.

function apiCall(name) {
    return new Promise(function(resolve) {
        setTimeout(function() {
            resolve(name + ": 200 Ok!");
        }, 1000);
    });
}

async function runSequential() {
    let r1 = await apiCall("Login");    // waits 1s
    let r2 = await apiCall("Dashboard"); // waits another 1s
    console.log(r1, r2); // Total: ~2s
}

147 - Parallel Execution

147_Parrallel_Execution.js Promise.all + await

When calls are independent, run them in parallel with Promise.all() + await for much faster execution — both calls start simultaneously instead of waiting for each other.

function apiCall(name) {
    return new Promise(function(resolve) {
        setTimeout(function() {
            resolve(name + ": 200 OK");
        }, 1000);
    });
}

async function runParallel() {
    let [r1, r2] = await Promise.all([
        apiCall("Users"),
        apiCall("Products")
    ]);
    console.log(r1, r2); // Total: ~1s (not 2s!)
}

149 - Real Flaky API Retry

149_API_REAL_FLAKY.js Retry Pattern

Real-world SDET pattern: retrying a flaky API call up to N times using a for loop with try/catch inside an async function. Exactly how Playwright's retries config works under the hood.

let attempt = 0;

function flakyAPI() {
    attempt++;
    if (attempt < 3) {
        return Promise.reject("Attempt " + attempt + ": failed");
    }
    return Promise.resolve("Attempt " + attempt + ": success!");
}

async function retryTesting(maxRetries) {
    for (let i = 1; i <= maxRetries; i++) {
        try {
            let result = await flakyAPI();
            console.log(result); return;
        } catch (err) {
            console.log(err);
        }
    }
}
