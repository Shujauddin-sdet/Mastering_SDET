150 - Export & Import

150_Export_import.js EXPORT_IMPORT ES Modules

Using import to bring in named exports from another module. Demonstrates importing BASE_URL and a utility function from testutil.js. Non-exported variables remain private to their module.

import { BASE_URL, formatUpperCaseString } from "../testutil.js";

console.log(BASE_URL); // "https://app.vwo.com"

let result = formatUpperCaseString("Pramod");
console.log(result); // "PRAMOD"

151 - Named Exports

testutil.js Named Export

Creating named exports with the export keyword. Only exported items are accessible from other modules — non-exported variables like fname stay private.

export let BASE_URL = "https://app.vwo.com";

export function formatUpperCaseString(sname) {
    return sname.toUpperCase();
}

let fname = "Pramod"; // NOT exported — private to this module

152 - Default Export

logger.js Default Export

A module can have ONE default export (the main thing it provides) plus additional named exports. Default exports can be imported with any name.

// Default Export → Export One Main Thing
export default function log(message) {
    console.log("[LOG] " + message);
}

export function log2(message) {
    console.log("[LOGS] " + message);
}

153 - Class Blueprint

153_Class_Objects.js CLASS_OBJECT CAB Pattern

A class is a blueprint: it defines Attributes (data) and Behaviour (methods). CAB = Class contains Attributes and Behaviour. Objects are instances created from this blueprint.

class Person {
    // Attributes
    name;
    email;
    salary;

    // Behaviour
    sleep() { }
    eat() { }
}

// CAB → Class contains Attribute, Behaviour

154 - Constructor

154_Car.js Constructor

The constructor() method runs when you create a new object with new. It initializes the object's properties using this to bind parameters to the instance.

class Car {
    constructor(assigned_name) {
        this.name = assigned_name;
    }
    drive() {
        console.log("Driving the car " + this.name);
    }
}

let hyndai_car = new Car("i10");
hyndai_car.drive(); // "Driving the car i10"

157 - APIClient IQ

157_IQ.js Multiple Instances

Interview-style question: creating multiple APIClient instances with different base URLs (staging vs production). Each object maintains its own independent state.

class APIClient {
    constructor(baseURL) {
        this.baseURL = baseURL;
    }
    get(path) {
        return this.baseURL + path;
    }
}

let staging = new APIClient("https://staging.api.com");
let prod = new APIClient("https://prod.api.com");
console.log(staging.get("/users"));
console.log(prod.get("/users"));

158 - Private & Public

158_Private_Public.js Private Fields (#)

JavaScript private fields use the # prefix. Private fields can only be accessed within the class — not from outside. Public fields have no prefix and are accessible anywhere.

class Credentials {
    #apiKey;
    user;

    constructor(user, key) {
        this.user = user;     // public
        this.#apiKey = key;   // private
    }
    pramodgetAuthHeader() {
        return "Bearer " + this.#apiKey;
    }
}

let cred = new Credentials("admin", "secret_key_1234");
console.log(cred.user);     // ✅ "admin"
// console.log(cred.#apiKey); // ❌ SyntaxError!

159 - Static Members

159_Static.js Static

Static properties and methods belong to the class itself, not to instances. Use ClassName.method() to access them — perfect for counters, summaries, or shared utilities.

class TestRunner {
    static totalTests = 0;
    static passCount = 0;

    constructor(name, passed) {
        this.name = name;
        TestRunner.totalTests++;
        if (passed) TestRunner.passCount++;
    }

    static summary() {
        return TestRunner.passCount + "/" + TestRunner.totalTests + " passed";
    }
}

160 - Static Part 2

160_Static_p2.js Shared Data

A Student class with a static collegeName shared across all instances. Static properties store data that belongs to the class, not individual objects — like a batch name shared by all students.

class Student {
    static collegeName = "PW AT Batch";

    constructor(name) {
        this.name = name;
    }
}

let amit = new Student("amit");
console.log(Student.collegeName); // "PW AT Batch" — class-level
console.log(amit.name);          // "amit" — instance-level

161 - Private Fields (Encapsulation)

161_Pramod_Child.js Encapsulation Getters & Setters

Encapsulation in action: hiding private fields (#child1, #child2) and exposing them only through getter and setter methods. This controls access and adds validation logic.

class Person {
    #child1;
    #child2;

    constructor(name, ch1, ch2) {
        this.name = name;
        this.#child1 = ch1;
        this.#child2 = ch2;
    }
    getChild1() { return this.#child1; }
    setChild1(changed_name) { this.#child1 = changed_name; }
}

162 - Car Encapsulation

162_Car.js Private Engine

A Car class where the #engine is private — you can only read or change it through getEngine() and setEngine() methods. This pattern protects internal state.

class Car {
    #engine;

    constructor(name, engineName) {
        this.name = name;
        this.#engine = engineName;
    }
    getEngine() { return this.#engine; }
    setEngine(nameEngine) { this.#engine = nameEngine; }
}

let tesla = new Car("Tesla", "V8");
tesla.setEngine("V9");
console.log(tesla.getEngine()); // "V9"

163 - Bank Account

163_Bank.js Access Control

Real-world encapsulation: an ICICI bank class with private #balance. The setter includes an authorization check (isCashier) — only authorized users can modify the balance.

class ICICI {
    #balance;

    constructor(name, balance) {
        this.#balance = balance;
        this.name = name;
    }
    getBalance() { return this.#balance; }
    setBalance(balance, isCashier) {
        if (isCashier) {
            this.#balance = balance;
        } else {
            console.log("Not allowed");
        }
    }
}
