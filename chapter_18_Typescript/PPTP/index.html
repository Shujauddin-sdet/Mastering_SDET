177 - Basic Types

177_Basic_Types.ts Primitive Types

All TypeScript primitive types: string, number, boolean, null, undefined. Plus typed arrays with both number[] and Array<string> syntax. And the escape hatches: any and unknown.

let name: string = "John";
let age: number = 30;
let pi: number = 3.14;
let isActive: boolean = true;
let nothing: null = null;
let notDefined: undefined = undefined;

// Arrays
let numbers: number[] = [1, 2, 3];
let names: Array<string> = ["John", "Jane"];

// Any (avoid!) and Unknown (safer)
let anything: any = "hello";

179 - Unknown & Functions

179_Unknown.ts unknown & Annotations

The unknown type is safer than any — you must type-check before using it. Also covers function annotations for parameters and return types, including arrow function syntax.

let unknown: unknown = "hello";

if (typeof unknown === "string") {
    console.log("It's a string!");
}

// Function annotations
function greet(name: string): string {
    return `Hello, ${name}!`;
}

// Arrow function annotations
const multiply = (a: number, b: number): number => a * b;

180 - void & never

180_Fn_NoReturn.ts void / never

void means a function returns nothing (like console.log). never means a function never returns at all — it either throws an error or runs an infinite loop.

// void — no return value
function sayHello(msg: string): void {
    console.log(msg);
}

// Function with return type
function greet(name: string): string {
    return `Hello, ${name}!`;
}

// never — function never returns
function throwError(message: string): never {
    throw new Error(message);
}

181 - Function IQ

181_IQ.ts Typed Functions

Real QA-flavored typed functions: buildEndpoint() returns a string, isSuccessCode() returns a boolean, and logTestStep() returns void. TypeScript enforces the contracts.

function buildEndpoint(base: string, path: string): string {
    return base + path;
}

function isSuccessCode(code: number): boolean {
    return code >= 200 && code < 300;
}

function logTestStep(step: string): void {
    console.log("[STEP] " + step);
}

console.log(buildEndpoint("https://api.com", "/users"));
console.log(isSuccessCode(200)); // true
console.log(isSuccessCode(404)); // false

183 - Filter Array

183_Filter_Array.ts Typed Filter

A fully typed function that filters HTTP response codes to find failures (≥ 400). Demonstrates typed parameters, return types, and typed callback functions with .filter().

let responseCodes: number[] = [200, 201, 404, 500, 302, 403];

function getFailedCodes(codes: number[]): number[] {
    return codes.filter(function(code: number): boolean {
        return code >= 400;
    });
}

console.log("All codes:", responseCodes);
console.log("Failed codes:", getFailedCodes(responseCodes));
// Failed codes: [404, 500, 403]
