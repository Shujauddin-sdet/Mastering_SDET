197 - Generic Functions

197_Generic.ts Generic <T>

Generics let you write one function that works with ANY type while still maintaining type safety. <T> is a type variable — it captures whatever type the caller provides.

function getFirstResults<T>(results: T[]): T {
    return results[0]!;
}

let firstCode = getFirstResults<number>([200, 404, 500]);
let firstTest = getFirstResults<string>(["Login", "Signup", "Cart"]);

console.log("First code:", firstCode); // 200
console.log("First test:", firstTest); // "Login"

198 - Generic Class

198_Generic_Class.ts Generic Class

A generic TestDataStorage<T> class that stores items of any type. One class works for numbers, strings, objects — with full type checking for each use case.

class TestDataStorage<T> {
    private items: T[] = [];

    add(item: T): void {
        this.items.push(item);
    }
    getFirst(): T {
        return this.items[0]!;
    }
    getAll(): T[] {
        return this.items;
    }
    count(): number {
        return this.items.length;
    }
}

let codes = new TestDataStorage<number>();
codes.add(200);
codes.add(404);

let names = new TestDataStorage<string>();
names.add("Login Test");

199 - Generic API Response

199_GENERIC_API_RESPOSNE.ts Wrapper Pattern

A generic wrapResponse<T> function that wraps any data type with a status code. One function handles string, boolean, and number responses — all type-safe.

function wrapResponse<T>(statusCode: number, data: T): { statusCode: number; data: T } {
    return { statusCode, data };
}

let userResp = wrapResponse<string>(200, "admin");
console.log(userResp); // { statusCode: 200, data: "admin" }

let flagResp = wrapResponse<boolean>(200, true);
console.log(flagResp); // { statusCode: 200, data: true }

let countResp = wrapResponse<number>(200, 42);
console.log(countResp); // { statusCode: 200, data: 42 }
